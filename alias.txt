ff() {
    repo_root="$(git rev-parse --show-toplevel)"
    current_dir="$(pwd)"
    if [[ "$repo_root" != "$current_dir" ]]; then
        echo "Not a the root of the repository ⚠️ "
        echo "Root path : $repo_root"
        echo "Curr Path : $current_dir"
        return 1
    fi
    git add .
    read "commit?Commit: "

    # Si le message est vide, on met un message par défaut
    if [[ -z "$commit" ]]; then
        commit="wip"
    fi

    git commit -m "$commit"
    git push
}

dx() {
   find . -type f -exec sed -i 's/docker-compose/docker compose/g' {} +
}

cl() {
    local current="$(pwd)"

    local protected_paths=(
        /
        /root
        /home
        /usr
        /bin
        /sbin
        /etc
        /var
        "$HOME"
    )

    for path in "${protected_paths[@]}"; do
        [[ "$current" == "$path" ]] && {
            echo "❌ Refusé : dossier protégé ($current)"
            return 1
        }
    done

    rm -rf -- * .[^.]*
}
